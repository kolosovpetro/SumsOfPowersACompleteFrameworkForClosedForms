The same method gives closed forms for double sums of powers. Lemma~\eqref{lem:centered-hockey-stick-identity} and Proposition~\eqref{prop:ordinary-sums-of-powers-decomposition} reduce the calculation to the sums
$\sum_{j=1}^{n} \binom{j -t + \frac{k}{2} +1}{k+1}$,
and $\sum_{j=1}^{n} \binom{j -t + \frac{k}{2}}{k+1}$.
Evaluating these sums gives the following proposition.
\begin{proposition}[Double sums of powers decomposition]
  \label{prop:double-sums-of-powers-decomposition}
  For non-negative integers $n,m$, and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{2}{n}{m} = \frac{1}{2} \sum_{k=0}^{m}
    \Bigg[
      &+ \binom{n -t + \frac{k}{2} +2}{k+2} - \binom{2 -t + \tfrac{k}{2}}{k+2} \KnuthRFoldSum{0}{n}{0} - \binom{1 -t + \tfrac{k}{2}}{k+1} \KnuthRFoldSum{1}{n}{0} \\
      &+ \binom{n -t + \frac{k}{2}+1}{k+2} - \binom{1-t + \tfrac{k}{2}}{k+2} \KnuthRFoldSum{0}{n}{0} - \binom{0-t + \tfrac{k}{2}}{k+1} \KnuthRFoldSum{1}{n}{0}
    \Bigg] \delta^{k} t^m.
  \end{align*}
\end{proposition}
